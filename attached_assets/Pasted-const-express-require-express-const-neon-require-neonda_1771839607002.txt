const express = require("express");
const { neon } = require("@neondatabase/serverless");

const sql = neon(process.env.DATABASE_URL);
const app = express();

const PORT = 3000;

/* ========= DASHBOARD PAGE ========= */
app.get("/mod", async (req, res) => {
  try {
    const start = Date.now();

    const logs = await sql`SELECT * FROM command_logs ORDER BY time DESC LIMIT 20`;
    const guilds = await sql`SELECT COUNT(*) FROM guilds`;
    const pingTest = await sql`SELECT NOW()`;

    const ping = Date.now() - start;

    const rows = logs.map(l => `
      <tr>
        <td>${l.username}</td>
        <td>${l.command}</td>
        <td>${l.channel_id}</td>
        <td>${new Date(l.time).toLocaleString()}</td>
      </tr>
    `).join("");

    res.send(`
      <html>
      <head>
        <title>Moderator Dashboard</title>
        <meta http-equiv="refresh" content="10">
        <style>
          body {
            background:#0f1116;
            color:#fff;
            font-family:Arial;
            padding:30px;
          }
          table {
            width:100%;
            border-collapse:collapse;
          }
          td, th {
            border:1px solid #333;
            padding:8px;
          }
          th {
            background:#1c1f26;
          }
          .card {
            background:#1c1f26;
            padding:15px;
            margin-bottom:20px;
            border-radius:8px;
          }
        </style>
      </head>

      <body>
        <h1>🛠 Bot Moderator Panel</h1>

        <div class="card">
          <b>Database Ping:</b> ${ping}ms<br>
          <b>Total Guilds:</b> ${guilds[0].count}
        </div>

        <h2>Recent Commands</h2>
        <table>
          <tr>
            <th>User</th>
            <th>Command</th>
            <th>Channel</th>
            <th>Time</th>
          </tr>
          ${rows}
        </table>
      </body>
      </html>
    `);

  } catch (err) {
    res.send("Dashboard error");
  }
});

app.listen(PORT, () => console.log("Dashboard running"));
