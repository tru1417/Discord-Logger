  // only mods/admins
  if (!message.member.permissions.has(PermissionFlagsBits.ManageGuild)) {
    return message.reply("❌ Moderator permission required.");
  }

  try {
    const start = Date.now();

    // quick health + stats queries
    const pingTest = await sql`SELECT NOW()`;
    const logs = await sql`SELECT COUNT(*) FROM command_logs`;
    const guilds = await sql`SELECT COUNT(*) FROM guilds`;

    const ping = Date.now() - start;

    const embed = new EmbedBuilder()
      .setTitle("🛠 Moderator Database Dashboard")
      .setDescription("Quick access to Neon control panel + live stats")
      .addFields(
        { name: "DB Ping", value: `${ping}ms`, inline: true },
        { name: "Command Logs", value: logs[0].count, inline: true },
        { name: "Guilds", value: guilds[0].count, inline: true },
        { name: "Open Neon Console", value: `[Click Here](${process.env.NEON_DASHBOARD})` }
      )
      .setFooter({ text: "Moderator only" })
      .setTimestamp();

    // send privately for safety
    await message.author.send({ embeds: [embed] });

    message.react("✅");

  } catch (err) {
    console.error(err);
    message.reply("❌ Dashboard failed to load.");
  }
});
